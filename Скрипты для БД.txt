Оригинальный скрипт

-- 1. Базовые таблицы
CREATE TABLE switches (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    type VARCHAR(100) NOT NULL,
    actuation_force INT,
    bottom_force INT,
    tactile_force INT,
    travel_length VARCHAR(50),
    price_per_switch DECIMAL(6, 2)
);

CREATE TABLE images (
    id SERIAL PRIMARY KEY,
    image_url VARCHAR(500) NOT NULL,
    alt_text VARCHAR(255)
);

-- 2. Основная таблица продуктов
CREATE TABLE products_demo (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    base_price DECIMAL(10, 2) NOT NULL,
    base_switch_id INT,
    switch_count DECIMAL(10, 2) DEFAULT 30.00,
    FOREIGN KEY (base_switch_id) REFERENCES switches(id)
);

-- 3. Зависимые таблицы
CREATE TABLE product_specifications (
    id SERIAL PRIMARY KEY,
    product_id INT,
    spec_key VARCHAR(255) NOT NULL,
    spec_value VARCHAR(500) NOT NULL,
    depends_on_switch BOOLEAN DEFAULT FALSE,
    display_order INT DEFAULT 0,
    FOREIGN KEY (product_id) REFERENCES products_demo(id) ON DELETE CASCADE
);

CREATE TABLE product_equipment (
    id SERIAL PRIMARY KEY,
    product_id INT,
    item_name VARCHAR(255) NOT NULL,
    display_order INT DEFAULT 0,
    FOREIGN KEY (product_id) REFERENCES products_demo(id) ON DELETE CASCADE
);

CREATE TABLE product_available_switches (
    id SERIAL PRIMARY KEY,
    product_id INT,
    switch_id INT,
    display_order INT DEFAULT 0,
    FOREIGN KEY (product_id) REFERENCES products_demo(id) ON DELETE CASCADE,
    FOREIGN KEY (switch_id) REFERENCES switches(id) ON DELETE CASCADE
);

CREATE TABLE product_images (
    id SERIAL PRIMARY KEY,
    product_id INT,
    image_id INT,
    sort_order INT DEFAULT 0,
    FOREIGN KEY (product_id) REFERENCES products_demo(id) ON DELETE CASCADE,
    FOREIGN KEY (image_id) REFERENCES images(id) ON DELETE CASCADE
);

-- Пользователи
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    login VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL,
    surname VARCHAR(255) NOT NULL,
    name VARCHAR(255) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Таблицы и поля для корзины, заказов и проверки остатков

-- Добавляем поле stock в products_demo (по умолчанию 10)
ALTER TABLE products_demo ADD COLUMN IF NOT EXISTS stock INT NOT NULL DEFAULT 10;

-- Добавляем роль пользователя для админки
ALTER TABLE users ADD COLUMN IF NOT EXISTS role VARCHAR(50) DEFAULT 'user';

-- Корзина
CREATE TABLE IF NOT EXISTS cart_items (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    session_token VARCHAR(255),
    product_id INT NOT NULL REFERENCES products_demo(id) ON DELETE CASCADE,
    switch_id INT REFERENCES switches(id) ON DELETE SET NULL,
    quantity INT NOT NULL CHECK (quantity > 0),
    unit_price DECIMAL(10,2) NOT NULL,
    added_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Заказы
CREATE TABLE IF NOT EXISTS orders (
    id SERIAL PRIMARY KEY,
    user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    total DECIMAL(10,2) NOT NULL,
    status VARCHAR(50) DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Состав заказа
CREATE TABLE IF NOT EXISTS order_items (
    id SERIAL PRIMARY KEY,
    order_id INT NOT NULL REFERENCES orders(id) ON DELETE CASCADE,
    product_id INT NOT NULL REFERENCES products_demo(id) ON DELETE CASCADE,
    switch_id INT REFERENCES switches(id) ON DELETE SET NULL,
    quantity INT NOT NULL,
    unit_price DECIMAL(10,2) NOT NULL
);
